  "Build targets that require SoapySDR (rf_adapt_intel, soapy_read_test)" ON)

option(BUILD_OFFLINE
  "Disable all FetchContent network access (requires dependency source trees to already exist locally, for example under build/_deps/ or via FETCHCONTENT_SOURCE_DIR_<name> overrides). Use on air-gapped nodes after a first online configure has populated the local FetchContent sources."
  OFF)

if(BUILD_OFFLINE)
  set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL
    "Populated by BUILD_OFFLINE — do not edit directly" FORCE)
  # Remember that we were the one who forced this, so the else() branch
  # can safely undo it without touching a user-supplied value.
  set(_MEEK_FC_DISCONNECTED_BY_BUILD_OFFLINE ON CACHE INTERNAL "")
  if(BUILD_HARDWARE_TARGETS)
    message(STATUS
      "BUILD_OFFLINE=ON: FetchContent will not access the network. "
      "Ensure the nlohmann_json and cpp_httplib source directories already "
      "exist locally (e.g. under build/_deps/) or set "
      "FETCHCONTENT_SOURCE_DIR_nlohmann_json and "
      "FETCHCONTENT_SOURCE_DIR_cpp_httplib.")
  else()
    message(STATUS
      "BUILD_OFFLINE=ON: FetchContent will not access the network. "
      "Ensure the nlohmann_json source directory already exists locally "
      "(e.g. under build/_deps/) or set "
      "FETCHCONTENT_SOURCE_DIR_nlohmann_json.")
  endif()
else()
  # Only undo the forced-ON value if BUILD_OFFLINE was the one that set it.
  # A user who passed -DFETCHCONTENT_FULLY_DISCONNECTED=ON independently is
  # left untouched.
  if(_MEEK_FC_DISCONNECTED_BY_BUILD_OFFLINE)
    unset(FETCHCONTENT_FULLY_DISCONNECTED CACHE)
    unset(_MEEK_FC_DISCONNECTED_BY_BUILD_OFFLINE CACHE)
  endif()
endif()

# ── Optional: liquid-dsp (FSK/PSK/OOK demodulation) ──────────────────────────
# Detection order: CMake config → pkg-config → find_library/find_path fallback.
# The compile definition HAVE_LIQUID is applied as a target-scoped definition
# on rf_adapt_intel (below) so that it appears in compile_commands.json entries
# for that target, enabling the CI verification step to confirm it is active.
find_package(liquid QUIET CONFIG)
if(liquid_FOUND)
  message(STATUS "Found liquid via CMake config: ${liquid_VERSION}")
  set(LIQUID_USE_CMAKE TRUE)
else()
  pkg_check_modules(LIQUID_PKG QUIET liquid)
  if(LIQUID_PKG_FOUND)
    message(STATUS "Found liquid via pkg-config: ${LIQUID_PKG_VERSION}")
  else()
    # Fallback: direct library/header search for source builds installed to
    # /usr/local that are not yet visible via pkg-config (e.g., PKG_CONFIG_PATH
    # not set, or no .pc file generated by the build system).
    find_library(LIQUID_FALLBACK_LIB NAMES liquid
      HINTS /usr/local/lib /usr/local/lib64
            /usr/local/lib/x86_64-linux-gnu /usr/lib/x86_64-linux-gnu)
    find_path(LIQUID_FALLBACK_INC NAMES liquid/liquid.h
      HINTS /usr/local/include /usr/include)
    if(LIQUID_FALLBACK_LIB AND LIQUID_FALLBACK_INC)
      message(STATUS "Found liquid via find_library: ${LIQUID_FALLBACK_LIB}")
      set(LIQUID_PKG_LIBRARIES "${LIQUID_FALLBACK_LIB}")
      set(LIQUID_PKG_INCLUDE_DIRS "${LIQUID_FALLBACK_INC}")
      set(LIQUID_PKG_FOUND TRUE)
    else()
      message(STATUS "liquid not found — advanced demod disabled")
    endif()
  endif()
endif()

# ── Include path shared by rf_adapt_intel, rf_audit, iq_metrics ──────────────
set(MEEK_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/include")

# ── Hardware-dependent targets ────────────────────────────────────────────────
if(BUILD_HARDWARE_TARGETS)

  # Probe for the C++20 library features used only by rf_adapt_intel.
  # cmake_push_check_state(RESET) clears only CMAKE_REQUIRED_* variables;
  # CMAKE_CXX_STANDARD is inherited so the try-compile uses the same standard
  # (-std=c++20 on GCC/Clang) without hard-coding a compiler-specific flag.
  # This catches mismatched compiler+stdlib combos regardless of compiler family.
  include(CMakePushCheckState)
  include(CheckCXXSourceCompiles)
  find_package(Threads REQUIRED)
  cmake_push_check_state(RESET)
  set(CMAKE_REQUIRED_LIBRARIES Threads::Threads)
  check_cxx_source_compiles([=[
#include <stop_token>
#include <thread>
#include <span>
int main() {
  std::jthread t([](std::stop_token) {});
  int arr[4]{};
  std::span<int> s(arr);
  (void)t; (void)s;
  return 0;
}
]=] _STDCXX_FEATURES_AVAILABLE)
  cmake_pop_check_state()
  if(NOT _STDCXX_FEATURES_AVAILABLE)
    # Provide targeted guidance for the most common failure: GCC < 12.
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU"
       AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS "12.0")
      message(FATAL_ERROR
        "C++${CMAKE_CXX_STANDARD} feature check failed: "
        "GCC ${CMAKE_CXX_COMPILER_VERSION} does not fully support the required "
        "standard library (std::jthread, std::stop_token, std::span in C++${CMAKE_CXX_STANDARD} mode).
"
        "GCC ≥ 12 is required.
"
        "On Bookworm the default GCC (gcc-12) satisfies this requirement.
"
        "If you only need the analysis tools (rf_audit, iq_metrics), you can skip the
"
        "hardware targets entirely:
"
        "  cmake -S . -B build -DBUILD_HARDWARE_TARGETS=OFF")
    else()
      message(FATAL_ERROR
        "C++${CMAKE_CXX_STANDARD} feature check failed "
        "(compiler: ${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}).\n"
        "The probe tests std::jthread, std::stop_token, and std::span — "
        "all required by the hardware targets.
"
        "Ensure your compiler and standard library support C++${CMAKE_CXX_STANDARD} "
        "(GCC ≥ 12 with libstdc++ ≥ 12, or Clang ≥ 14 with a compatible libc++).
"
        "If you only need the analysis tools (rf_audit, iq_metrics), configure with:
"
        "  -DBUILD_HARDWARE_TARGETS=OFF")
    endif()
  endif()

  pkg_check_modules(SOAPY_PKG QUIET SoapySDR)
  if(NOT SOAPY_PKG_FOUND)
    message(FATAL_ERROR
      "SoapySDR not found.  Install libsoapysdr-dev or set "
      "BUILD_HARDWARE_TARGETS=OFF to skip hardware targets.")
  endif()

  pkg_check_modules(SQLITE3 REQUIRED sqlite3)

  # cpp-httplib (optional Prometheus HTTP server)
  FetchContent_Declare(
    cpp_httplib
    URL https://github.com/yhirose/cpp-httplib/archive/refs/tags/v0.18.3.tar.gz
    URL_HASH SHA256=a0567bcd6c3fe5cef1b329b96245119047f876b49e06cc129a36a7a8dffe173e
  )
  set(HTTPLIB_COMPILE OFF CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(cpp_httplib)

  # ── rf_adapt_intel daemon ─────────────────────────────────────────────────
  add_executable(rf_adapt_intel src/main.cpp)

  target_include_directories(rf_adapt_intel PRIVATE
    ${MEEK_INCLUDE_DIR}
    ${SOAPY_PKG_INCLUDE_DIRS}
    ${SQLITE3_INCLUDE_DIRS}
  )

  target_link_libraries(rf_adapt_intel PRIVATE
    ${SOAPY_PKG_LIBRARIES}
    ${SQLITE3_LIBRARIES}
    Threads::Threads
    nlohmann_json::nlohmann_json
  )

  if(TARGET httplib::httplib)
    target_link_libraries(rf_adapt_intel PRIVATE httplib::httplib)
    target_compile_definitions(rf_adapt_intel PRIVATE HAVE_HTTPLIB)
  endif()

  # Wire liquid-dsp into rf_adapt_intel and apply HAVE_LIQUID as a
  # target-scoped definition so it appears in compile_commands.json.
  if(LIQUID_USE_CMAKE)
    target_link_libraries(rf_adapt_intel PRIVATE liquid::liquid)
    target_compile_definitions(rf_adapt_intel PRIVATE HAVE_LIQUID)
  elseif(LIQUID_PKG_FOUND)
    target_include_directories(rf_adapt_intel PRIVATE ${LIQUID_PKG_INCLUDE_DIRS})
    target_link_libraries(rf_adapt_intel PRIVATE ${LIQUID_PKG_LIBRARIES})
    target_compile_definitions(rf_adapt_intel PRIVATE HAVE_LIQUID)
  endif()

  # Optional: libsystemd for sd_notify watchdog integration
  pkg_check_modules(LIBSYSTEMD QUIET libsystemd)
  if(LIBSYSTEMD_FOUND)
    message(STATUS "libsystemd found — READY/WATCHDOG/STOPPING all use native sd_notify "
                   "via <systemd/sd-daemon.h>")
    target_compile_definitions(rf_adapt_intel PRIVATE HAVE_SYSTEMD)
    target_link_libraries(rf_adapt_intel PRIVATE ${LIBSYSTEMD_LIBRARIES})
    target_include_directories(rf_adapt_intel PRIVATE ${LIBSYSTEMD_INCLUDE_DIRS})
  else()
    message(STATUS "libsystemd not found — READY/WATCHDOG/STOPPING all use socket-based "
                   "sd_notify fallback (NOTIFY_SOCKET datagram)")
  endif()

  target_compile_options(rf_adapt_intel PRIVATE
    -Wall -Wextra -Wno-unused-parameter)
  # HAVE_SOAPY enables the SoapySdrSource implementation in isdr_source.hpp
  target_compile_definitions(rf_adapt_intel PRIVATE HAVE_SOAPY)

  # ── soapy_read_test ───────────────────────────────────────────────────────
  add_executable(soapy_read_test tools/soapy_read_test.cpp)
  target_include_directories(soapy_read_test PRIVATE ${SOAPY_PKG_INCLUDE_DIRS})
  target_link_libraries(soapy_read_test PRIVATE ${SOAPY_PKG_LIBRARIES} Threads::Threads)
  target_compile_options(soapy_read_test PRIVATE
    -Wall -Wextra -Wno-unused-parameter)

  install(TARGETS rf_adapt_intel
    RUNTIME DESTINATION bin
    PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE
                GROUP_READ GROUP_EXECUTE
                WORLD_READ WORLD_EXECUTE
  )

else()
  message(STATUS
    "BUILD_HARDWARE_TARGETS=OFF: skipping rf_adapt_intel and soapy_read_test.")
endif()
